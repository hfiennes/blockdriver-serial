480.entry
490      stmfd   r13!,{r14}
500      cmp     r0,#((endoftable-table)/4)
510      addcc   pc,pc,r0,lsl#2
520      ldmfd   r13!,{pc}^
530
540.table
550      bal     put_byte
560      bal     get_byte
570      bal     put_block
580      bal     get_block
590      bal     check_tx
600      bal     check_rx
610      bal     flush_tx
620      bal     flush_rx
630      bal     control_lines
640      bal     read_modem_control
650      bal     read_rx_errors
660      bal     send_break
670      bal     examine_byte
680      bal     speed
690      bal     speed
700      bal     wordformat
710      bal     flowcontrol
720      bal     initialise
730      bal     closedown
740.endoftable
750
760      FNcheck_and_set(&80)
770.driverinfo
780      equs    drtitle$+z$
790
800      FNcheck_and_set(&A0)
810.manufacturerinfo
820      equs    "The Serial Port"+z$
830
840      FNcheck_and_set(&C0)
850.version
860      equd    &0001000e     ; v1.14
870.flags
880      equd    flag_morethanone + flag_hardwarefifo + flag_controlbreak + flag_supportsblock + flag_supportsinq + (23<<24)
890.drivernumber
900      equd    realdriver_number<<8
910
920      FNcheck_and_set(&100)
930.speeds
940      equd        50
950      equd        75
960      equd       110
970      equd       134
980      equd       150
990      equd       300
1000      equd       600
1010      equd      1200
1020      equd      1800
1030      equd      2400
1040      equd      3600
1050      equd      4800
1060      equd      7200
1070      equd      9600
1080      equd     14400
1090      equd     19200
1100      equd     38400
1110      equd     57600
1120.maxissue0
1130      equd    115200
1140.maxissue1
1150      equd    230400
1160.maxissue2
1170      equd    460800
1180      equd         0
1190
1200      FNcheck_and_set(&180)
1210;--- Put byte routine ---------------------------------------------------
1220.put_byte
1230      and     r0,r1,#&ff
1240      and     r1,r2,#&ff
1250      swi     "XSPSerial_TX"
1260      movcc   r0,#0
1270      mvncs   r0,#0
1280      ldmfd   r13!,{pc}^
1290
1300;--- Get byte routine ---------------------------------------------------
1310.get_byte
1320      and     r0,r1,#&ff
1330      mov     r1,#0
1340      swi     "XSPSerial_RX"
1350      mov     r0,r1
1360      ldmfd   r13!,{pc}^
1370
1380;--- Put block ----------------------------------------------------------
1390.put_block
1400      cmp     r3,#0
1410      beq     put_block_1        ; Workaround for bug in firmware
1420      and     r0,r1,#&ff
1430      mov     r1,r2
1440      mov     r2,r3
1450      swi     "XSPSerial_TX"
1460      sub     r0,r3,r2
1470      ldmfd   r13!,{pc}^
1480
1490.put_block_1
1500      mov     r0,#0
1510      ldmfd   r13!,{pc}^
1520
1530;--- Get block ----------------------------------------------------------
1540.get_block
1550      cmp     r3,#0
1560      beq     get_block_1        ; Workaround for bug in firmware
1570      and     r0,r1,#&ff
1580      mov     r1,r2
1590      mov     r2,r3
1600      swi     "XSPSerial_RX"
1610      sub     r0,r3,r2
1620      ldmfd   r13!,{pc}^
1630
1640.get_block_1
1650      mov     r0,#0
1660      ldmfd   r13!,{pc}^
1670
1680;--- Check TX buffer ----------------------------------------------------
1690.check_tx
1700      and     r0,r1,#&ff
1710      swi     "XSPSerial_TXCount"
1720      mov     r0,r2
1730      ldmfd   r13!,{pc}^
1740
1750;--- Check RX buffer ----------------------------------------------------
1760.check_rx
1770      and     r0,r1,#&ff
1780      swi     "XSPSerial_RXCount"
1790      mov     r0,r1
1800      ldmfd   r13!,{pc}^
1810
1820;--- Flush TX buffer ----------------------------------------------------
1830.flush_tx
1840      and     r0,r1,#&ff
1850      swi     "XSPSerial_TXPurge"
1860      ldmfd   r13!,{pc}^
1870
1880;--- Flush RX buffer ----------------------------------------------------
1890.flush_rx
1900      and     r0,r1,#&ff
1910      swi     "XSPSerial_RXPurge"
1920      ldmfd   r13!,{pc}^
1930
1940;--- Control lines ------------------------------------------------------
1950.control_lines
1960      stmfd   r13!,{r1}
1970      cmp     r2,#0
1980      blt     control_lines_read
1990
2000      and     r0,r1,#&ff
2010      and     r1,r2,#7
2020      mvn     r2,#7
2030      swi     "XSPSerial_Control"
2040
2050.control_lines_read
2060      ldmfd   r13!,{r0}
2070      and     r0,r0,#&ff
2080      mov     r1,#0
2090      mvn     r2,#0
2100      swi     "XSPSerial_Control"
2110      and     r0,r1,#7
2120
2130      ldmfd   r13!,{pc}^
2140
2150;--- Read modem control lines -------------------------------------------
2160.read_modem_control
2170      and     r0,r1,#&ff
2180      mov     r1,#0
2190      mvn     r2,#0
2200      swi     "XSPSerial_Control"
2210      mov     r0,r1,lsr#16
2220      and     r0,r0,#&f
2230
2240      ldmfd   r13!,{pc}^
2250
2260;--- Read RX errors -----------------------------------------------------
2270.read_rx_errors
2280      mov     r0,#0
2290      ldmfd   r13!,{pc}^
2300
2310;--- Send break ---------------------------------------------------------
2320.send_break
2330      and     r1,r1,#&ff
2340      stmfd   r13!,{r1}
2350      cmp     r2,#0
2360      ble     send_break_exit
2370      
2380      swi     "XOS_ReadMonotonicTime"
2390      add     r3,r0,r2  ; make break end time
2400
2410      mov     r0,r1
2420      mov     r1,#4      ; set break
2430      mvn     r2,#4      ; everything but break
2440      swi     "XSPSerial_Control"
2450      
2460.send_break_loop
2470      swi     "XOS_ReadMonotonicTime"
2480      cmp     r0,r3
2490      blt     send_break_loop
2500      
2510.send_break_exit
2520      ldmfd   r13!,{r0}
2530      mov     r1,#0      ; clear break
2540      mvn     r2,#4      ; everything but (the) break
2550      swi     "XSPSerial_Control"
2560      ldmfd   r13!,{pc}^
2570
2580;--- Examine byte -------------------------------------------------------
2590.examine_byte
2600      and     r0,r1,#&ff
2610      mov     r1,#0
2620      swi     "XSPSerial_RXPeek"
2630      mov     r0,r1
2640      ldmfd   r13!,{pc}^
2650
2660;--- Set speed ----------------------------------------------------------
2670.speed
2680      stmfd   r13!,{r1}
2690      and     r0,r1,#&ff
2700      cmp     r2,#0
2710      blt     speed_read
2720
2730      mov     r1,r2
2740      swi     "XSPSerial_Speed"
2750
2760.speed_read
2770      ldmfd   r13!,{r0}
2780      and     r0,r0,#&ff
2790      mvn     r1,#0
2800      swi     "XSPSerial_Speed"
2810      mov     r0,r1
2820      ldmfd   r13!,{pc}^
2830
2840;--- Set word format ----------------------------------------------------
2850.wordformat
2860      stmfd   r13!,{r1}
2870      and     r0,r1,#&ff
2880      cmp     r2,#0
2890      blt     wordformat_read
2900
2910      mov     r1,r2
2920      swi     "XSPSerial_Format"
2930
2940.wordformat_read
2950      ldmfd   r13!,{r0}
2960      and     r0,r0,#&ff
2970      mvn     r1,#0
2980      swi     "XSPSerial_Format"
2990      mov     r0,r1
3000      ldmfd   r13!,{pc}^
3010
3020;--- Set flow control ---------------------------------------------------
3030.flowcontrol
3040      and     r0,r1,#&ff
3050      cmp     r2,#0
3060      blt     flowcontrol_read
3070
3080      cmp     r2,#0
3090      moveq   r1,#0         ; No tx/rx flow control
3100      cmp     r2,#1
3110      moveq   r1,#%00101000 ; tx & rx hardware
3120      cmp     r2,#2
3130      moveq   r1,#%01010000 ; tx & rx xon/xoff
3140      cmp     r2,#3
3150      moveq   r1,#%01111000 ; tx & rx both
3160      str     r2,flowcontrol_last
3170
3180      ldr     r2,flowcontrol_mask
3190      swi     "XSPSerial_Control"
3200
3210.flowcontrol_read
3220      ldr     r0,flowcontrol_last
3230      ldmfd   r13!,{pc}^
3240
3250.flowcontrol_mask
3260      equd    &fff87
3270
3280.flowcontrol_last
3290      equd    1
3300
3310;--- Initialise driver --------------------------------------------------
3320.initialise
3330      mov     r3,r1           ; save r1
3340
3350      ; Check number of ports
3360      mvn     r2,#0 ; -1
3370
3380.initialise_0
3381      add     r2,r2,#1
3390      mov     r0,r2
3410      mvn     r1,#0 ; -1
3420      swi     "XSPSerial_Format"
3430      bvs     initialise_1
3440
3450      cmn     r1,#1
3460      beq     initialise_1   ; still -1? not a valid port
3470
3480      bal     initialise_0
3490
3500.initialise_1
3510      sub     r2,r2,#1
3520      strb    r2,(flags+3)
3530
3540      cmp     r2,#0          ; any ports found at all?
3550      ble     initialise_err
3560
3570      swi     "XSPSerial_Info"
3580      ldrvs   r1,maxissue0
3590      mov     r2,#0
3600      
3610      ldr     r0,maxissue1   ; Fixup baudrate table
3620      cmp     r1,r0
3630      strlt   r2,maxissue1
3640      
3650      ldr     r0,maxissue2
3660      cmp     r1,r0
3670      strlt   r2,maxissue2
3680      
3690      cmn     r3,#1
3700      moveq   r0,#0
3710      ldmeqfd r13!,{pc}^
3720
3730      movs    r0,#0 ; clear flags
3740
3750      ; Default - rx/tx cts flow, dtr & rts active
3760      and     r0,r3,#&ff
3770      mov     r1,#initbitset
3780      ldr     r2,initmask
3790      swi     "XSPSerial_Control"
3800
3810      movvc   r0,#0
3820      ldmvcfd r13!,{pc}^
3830
3840.initialise_err
3850      adr     r0,init_error
3860      ldmfd   r13!,{pc}^
3870
3880.init_error
3890      equs    "No Single/Dual/Triple Serial card installed"+CHR$0
3900      align
3910
3920.initmask
3930      equd    &ffffff00
3940
3950;--- Close down driver --------------------------------------------------
3960.closedown
3970      ldmfd   r13!,{pc}^
3980]
3990NEXT
