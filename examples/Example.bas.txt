10DIM driver 4096
20
30OSCLI("LOAD SerialDev:modules.internal.driver "+STR$~driver)
40
50DRIVER_PUTBYTE       =0
60DRIVER_GETBYTE       =1
70DRIVER_PUTBLOCK      =2
80DRIVER_GETBLOCK      =3
90DRIVER_CHECKTX       =4
100DRIVER_CHECKRX       =5
110DRIVER_FLUSHTX       =6
120DRIVER_FLUSHRX       =7
130DRIVER_CONTROLLINES  =8
140DRIVER_MODEMCONTROL  =9
150DRIVER_RXERRORS      =10
160DRIVER_BREAK         =11
170DRIVER_EXAMINE       =12
180DRIVER_TXSPEED       =13
190DRIVER_RXSPEED       =14
200DRIVER_WORDFORMAT    =15
210DRIVER_FLOWCONTROL   =16
220DRIVER_INITIALISE    =17
230DRIVER_CLOSEDOWN     =18
240DRIVER_POLL          =19
250
260J%=FNd(DRIVER_INITIALISE,0,0,0)
270IF J%<>0 THEN
280  err$=""
290  REPEAT
300    err$+=CHR$?J%
310    J%+=1
320  UNTIL ?J%=0
330  PRINT "Error when initialising driver: ";err$
340  END
350ENDIF
360J%=FNd(DRIVER_TXSPEED,0,19200,0)
370J%=FNd(DRIVER_RXSPEED,0,19200,0)
380J%=FNd(DRIVER_WORDFORMAT,0,0,0)
390
400REM Drop DTR & RTS for 1 second
410J%=FNd(DRIVER_CONTROLLINES,0,0,0)
420T%=TIME
430REPEATJ%=FNd(DRIVER_POLL,0,0,0):UNTIL(TIME-T%)>100
440J%=FNd(DRIVER_CONTROLLINES,0,3,0)
450
460REM Main loop
470REPEAT
480  J%=FNd(DRIVER_POLL,0,0,0)
490  IF FNd(DRIVER_CHECKRX,0,0,0)>0 THEN
500    J%=FNd(DRIVER_GETBYTE,0,0,0)
510    IF J%>31 OR J%=10 OR J%=13 VDU J%
520  ENDIF
530  IF ADVAL(-1)>0 THEN
540    J%=FNd(DRIVER_PUTBYTE,0,GET,0)
550  ENDIF
560UNTIL 0
570
580END
590
600DEFFNd(A%,B%,C%,D%)
610:=USR(driver)
